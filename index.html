window.onload = function() { // Só faz auto-login se houver promotor válido salvo try { var pSalvo = localStorage.getItem('univalidade_promotor'); if (pSalvo) { var p = JSON.parse(pSalvo); if (p && p.nome && p.loja && p.cidade) { promotor = p; document.getElementById('inNome').value = p.nome; document.getElementById('inLoja').value = p.loja; document.getElementById('inCidade').value = p.cidade; entrar(); return; } } } catch(e) {} // Se não houver promotor válido, garante que a tela de login apareça document.getElementById('telaLogin').classList.add('ativa'); document.getElementById('telaDash').classList.remove('ativa'); document.getElementById('fab').style.display = 'none'; document.getElementById('nav').style.display = 'none'; }; function entrar() { var nome = document.getElementById('inNome').value.trim(); var loja = document.getElementById('inLoja').value.trim(); var cidade = document.getElementById('inCidade').value.trim(); if (!nome || !loja || !cidade) { alert('Preencha todos os campos!'); return; } // Reset de estado interno antes de montar a tela promotor = { nome: nome, loja: loja, cidade: cidade }; localStorage.setItem('univalidade_promotor', JSON.stringify(promotor)); document.getElementById('saudNome').textContent = nome.split(' ')[0]; document.getElementById('saudLoja').textContent = loja + ' • ' + cidade; document.getElementById('telaLogin').classList.remove('ativa'); document.getElementById('telaDash').classList.add('ativa'); document.getElementById('fab').style.display = 'flex'; document.getElementById('nav').style.display = 'flex'; renderizar(); } function sair() { // Limpa tudo antes de recarregar try { localStorage.removeItem('univalidade_promotor'); } catch(e) {} promotor = null; produtos = []; categoriaSelecionada = null; produtoDetalheId = null; // Fecha modais abertos, se houver var modalBg = document.getElementById('modalBg'); var modalDetalheBg = document.getElementById('modalDetalheBg'); if (modalBg) modalBg.classList.remove('aberto'); if (modalDetalheBg) modalDetalheBg.classList.remove('aberto'); // Recarrega com parâmetro para evitar cache do bfcache location.replace(location.pathname + '?logout=' + Date.now()); }
